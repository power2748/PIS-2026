```python
# services/order_service/infrastructure/adapter/out/rabbitmq_event_publisher.py
import json, pika, os
from dataclasses import asdict
from application.port.out.event_publisher import IEventPublisher
 
ROUTING_KEYS = {
    "OrderCreatedEvent":       "order.created",
    "OrderSentToKitchenEvent": "order.sent_to_kitchen",
    "OrderCancelledEvent":     "order.cancelled",
}
 
 
class RabbitMQEventPublisher(IEventPublisher):
    """
    Публикует доменные события Order Service в RabbitMQ.
    Exchange: restaurant.events (topic, durable).
    """
    EXCHANGE = "restaurant.events"
 
    def __init__(self):
        self._url = os.getenv("RABBITMQ_URL", "amqp://guest:guest@rabbitmq:5672/")
 
    def publish(self, event) -> None:
        key = ROUTING_KEYS.get(type(event).__name__)
        if not key:
            return
        try:
            conn = pika.BlockingConnection(pika.URLParameters(self._url))
            ch = conn.channel()
            ch.exchange_declare(exchange=self.EXCHANGE, exchange_type="topic", durable=True)
            payload = json.dumps({
                **{k: str(v) for k, v in vars(event).items()},
                "_event_type": type(event).__name__,
            })
            ch.basic_publish(
                exchange=self.EXCHANGE,
                routing_key=key,
                body=payload.encode(),
                properties=pika.BasicProperties(delivery_mode=2),
            )
            conn.close()
        except Exception as e:
            print(f"[OrderService][Publisher] WARNING: {key} not published: {e}")
```

```python
# services/kitchen_service/infrastructure/adapter/out/rabbitmq_consumer.py
import json, pika, os, threading
from application.command.create_ticket_command import CreateTicketCommand
from application.command.handlers.create_ticket_handler import CreateTicketHandler
from infrastructure.config.dependency_injection import build_create_ticket_handler
 
 
class KitchenEventConsumer:
    """
    Слушает события из Order Service и создаёт тикеты кухни.
    Подписан на: order.sent_to_kitchen, order.cancelled.
    """
    EXCHANGE = "restaurant.events"
    QUEUE    = "kitchen-service"
 
    def __init__(self):
        self._url = os.getenv("RABBITMQ_URL", "amqp://guest:guest@rabbitmq:5672/")
 
    def start(self):
        thread = threading.Thread(target=self._consume, daemon=True)
        thread.start()
 
    def _consume(self):
        conn = pika.BlockingConnection(pika.URLParameters(self._url))
        ch = conn.channel()
        ch.exchange_declare(exchange=self.EXCHANGE, exchange_type="topic", durable=True)
        ch.queue_declare(queue=self.QUEUE, durable=True)
        ch.queue_bind(exchange=self.EXCHANGE, queue=self.QUEUE, routing_key="order.*")
        ch.basic_qos(prefetch_count=1)
        ch.basic_consume(queue=self.QUEUE, on_message_callback=self._handle)
        print("[KitchenConsumer] Listening for events...")
        ch.start_consuming()
 
    def _handle(self, ch, method, props, body):
        try:
            data = json.loads(body)
            event_type   = data.get("_event_type")
            routing_key  = method.routing_key
 
            if routing_key == "order.sent_to_kitchen":
                handler: CreateTicketHandler = build_create_ticket_handler()
                command = CreateTicketCommand(
                    ticket_id=data["ticket_id"],
                    order_id=data["order_id"],
                    table_number=int(data["table_number"]),
                    items=data.get("items", []),
                )
                handler.handle(command)
 
            elif routing_key == "order.cancelled":
                # отменяем тикет если есть
                from infrastructure.config.dependency_injection import build_cancel_ticket_handler
                build_cancel_ticket_handler().handle(data["order_id"])
 
            ch.basic_ack(method.delivery_tag)
 
        except Exception as e:
            print(f"[KitchenConsumer] ERROR: {e}")
            ch.basic_nack(method.delivery_tag, requeue=False)
```

### 4. Notification Service
 
**Bounded Context:** Доставка уведомлений — подписывается на события всех остальных сервисов и отправляет push на KDS-дисплеи и официантам.


```python
    "order.created":          "_on_order_created",
    "order.sent_to_kitchen":  "_on_order_sent_to_kitchen",
    "payment.completed":      "_on_payment_completed",
    "payment.failed":         "_on_payment_failed",
}
 
 
class NotificationConsumer:
    """
    Notification Service — конечная точка событийной цепочки.
    Не публикует события, только потребляет и доставляет уведомления.
    """
    EXCHANGE = "restaurant.events"
    QUEUE    = "notification-service"
 
    def __init__(self):
        self._url = os.getenv("RABBITMQ_URL", "amqp://guest:guest@rabbitmq:5672/")
 
    def start(self):
        thread = threading.Thread(target=self._run, daemon=True)
        thread.start()
 
    def _run(self):
        conn = pika.BlockingConnection(pika.URLParameters(self._url))
        ch = conn.channel()
        ch.exchange_declare(exchange=self.EXCHANGE, exchange_type="topic", durable=True)
        ch.queue_declare(queue=self.QUEUE, durable=True)
        for key in HANDLERS:
            ch.queue_bind(exchange=self.EXCHANGE, queue=self.QUEUE, routing_key=key)
        ch.basic_qos(prefetch_count=5)
        ch.basic_consume(queue=self.QUEUE, on_message_callback=self._handle)
        ch.start_consuming()
```

### 7. Circuit Breaker
 
```python
# shared/circuit_breaker.py
import time
from enum import Enum
from typing import Callable, Any
 
 
class CBState(Enum):
    CLOSED    = "CLOSED"     # нормальная работа
    OPEN      = "OPEN"       # все запросы блокируются
    HALF_OPEN = "HALF_OPEN"  # пробная попытка
 
 
class CircuitBreaker:
    """
    Circuit Breaker для межсервисных HTTP-вызовов.
    CLOSED → OPEN при N ошибках за окно времени.
    OPEN → HALF_OPEN через timeout_seconds.
    HALF_OPEN → CLOSED при успехе, → OPEN при ошибке.
    """
 
    def __init__(
        self,
        name: str,
        failure_threshold: int = 5,
        success_threshold: int = 2,
        timeout_seconds: float = 30.0,
    ):
        self.name              = name
        self._failure_threshold = failure_threshold
        self._success_threshold = success_threshold
        self._timeout           = timeout_seconds
        self._state             = CBState.CLOSED
        self._failure_count     = 0
        self._success_count     = 0
        self._last_failure_time: float = 0
 
    def call(self, func: Callable, *args, fallback=None, **kwargs) -> Any:
        if self._state == CBState.OPEN:
            if time.time() - self._last_failure_time >= self._timeout:
                self._state = CBState.HALF_OPEN
                self._success_count = 0
                print(f"[CB:{self.name}] OPEN → HALF_OPEN")
            else:
                print(f"[CB:{self.name}] OPEN — запрос отклонён")
                if fallback is not None:
                    return fallback()
                raise CircuitBreakerOpenError(f"Circuit breaker {self.name} is OPEN")
 
        try:
            result = func(*args, **kwargs)
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise
 
    def _on_success(self):
        if self._state == CBState.HALF_OPEN:
            self._success_count += 1
            if self._success_count >= self._success_threshold:
                self._state = CBState.CLOSED
                self._failure_count = 0
                print(f"[CB:{self.name}] HALF_OPEN → CLOSED")
        elif self._state == CBState.CLOSED:
            self._failure_count = 0
 
    def _on_failure(self):
        self._failure_count    += 1
        self._last_failure_time = time.time()
        if self._state == CBState.HALF_OPEN:
            self._state = CBState.OPEN
            print(f"[CB:{self.name}] HALF_OPEN → OPEN")
        elif self._failure_count >= self._failure_threshold:
            self._state = CBState.OPEN
            print(f"[CB:{self.name}] CLOSED → OPEN ({self._failure_count} failures)")
 
    @property
    def state(self) -> CBState:
        return self._state
 
 
class CircuitBreakerOpenError(Exception):
    pass
```

## Вывод
 
Монолит разбит на 4 микросервиса по принципу bounded contexts: Order Service владеет заказами и столиками, Kitchen Service — тикетами и станциями, Payment Service — платежами и эквайрингом, Notification Service — доставкой уведомлений. Каждый сервис имеет собственную БД (Database-per-Service) и не обращается напрямую к чужим таблицам.
 
Коммуникация — полностью асинхронная через RabbitMQ topic exchange `restaurant.events`. Один сервис публикует событие, ноль или несколько других подписываются — низкая связанность, сервисы не знают друг о друге. Circuit Breaker защищает Payment Service от каскадного падения при недоступности Stripe: после 3 ошибок подряд запросы блокируются на 60 секунд, затем делается пробная попытка.
 
Ключевой компромисс микросервисной архитектуры — eventual consistency: Kitchen Service не получает тикет мгновенно, а только после обработки события из очереди. Зато сбой Kitchen Service не влияет на создание заказов в Order Service.
